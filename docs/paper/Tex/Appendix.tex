\chapter{群论归约：最坏排列模式的代表元}\label{app:symmetry}

本附录给出第~\ref{sec:ov-performance}节引用的群论归约的完整推导。
需要强调：这是把 $|\mathcal{R}|$ 从 $n!$ 压到可控规模的\textbf{实现技巧}，
不改变约束形式，也\textbf{非本文核心贡献}——列此仅供读者核验
第~\ref{sec:ov-performance}节「排列模式集合 $\mathcal{R}$」的规模依据。

\section{问题定义}

设物理拓扑为无向图 $H=(V,E)$，$|V|=n$。负载均衡策略 $L$ 把端到端流量需求矩阵
$T=(t_{ij})_{n\times n}$ 映射为流图 $G(T)=L(H,T)$，瓶颈负载
$\Phi(T)=\max_{e\in E}\ell_e\bigl(G(T)\bigr)$。
需求矩阵满足\textbf{双随机}（每行和、每列和均为 1，对应每个端口满速率同时收发），
其全体构成 Birkhoff 多面体 $\mathcal{T}$。我们不知 $T$，需求最坏情形下的瓶颈负载：

\begin{equation}\label{eq:sym-phi}
    T^* \;=\; \arg\max_{T\in\mathcal{T}}\ \Phi(T)
\end{equation}

若负载均衡为最优（自适应/最优多商品流），$\Phi(T)$ 本身是并发流的最优值，
外层 $\max_T\Phi(T)$ 是凸函数最大化——一般情形无多项式保证。下面的归约正是把这一困难压掉。

\section{定理 1：最坏模式为置换矩阵}

\textbf{定理 1。}
设 $H$ 顶点传递（vertex-transitive）且负载均衡策略 $L$ 在 $\operatorname{Aut}(H)$ 作用下不变，
则存在最坏流量模式 $T^*$ 使得每行恰有一个元素非零，即 $T^*$ 为\textbf{置换矩阵}。

\textit{证明思路。}
$\Phi:\mathcal{T}\to\mathbb{R}^+$ 为凸函数；$\mathcal{T}$ 为 Birkhoff 多面体，
其顶点恰为置换矩阵（Birkhoff--von Neumann 定理~\cite{birkhoff1946tres}）；
凸函数在有界多面体上的最大值必在某个顶点处取到。
顶点传递性与 $L$ 的 $\operatorname{Aut}(H)$-不变性保证 $\Phi$ 限制在 $\mathcal{T}$ 上确为凸。$\square$

\textbf{推论。}只需考虑 $\sigma\in S_n$ 对应的流量模式：

\begin{equation}
    t_{ij}^{(\sigma)} \;=\; \begin{cases} 1, & j=\sigma(i),\\ 0, & \text{否则},\end{cases}
\end{equation}

至此搜索空间从 $n^2$ 维连续多面体降为 $n!$ 个离散点。

\section{定理 2：共轭归约}

设 $L$ 是 $\operatorname{Aut}(H)$-不变的：

\begin{equation}\label{eq:sym-A1}
    \forall \varphi\in\operatorname{Aut}(H),\qquad
    L\bigl(H,\ \varphi\circ T\circ\varphi^{-1}\bigr) \;=\; \varphi\cdot L(H,T),
\end{equation}

其中 $(\varphi\circ T\circ\varphi^{-1})_{ij}=T_{\varphi^{-1}(i),\,\varphi^{-1}(j)}$。

\textbf{引理 1（节点重标号 = 置换共轭）。}
按 $\varphi$ 对节点重标号后，置换 $\sigma$ 变为 $\varphi\sigma\varphi^{-1}$。

\textit{证明。}原流量 $i\to\sigma(i)$，重标号为 $\varphi(i)\to\varphi(\sigma(i))$；
对新标号 $j=\varphi(i)$，有 $j\mapsto\varphi(\sigma(\varphi^{-1}(j)))=(\varphi\sigma\varphi^{-1})(j)$。$\square$

\textbf{引理 2（流图同构 = 置换共轭）。}
在式~\eqref{eq:sym-A1} 下，若 $\sigma\mapsto G_\sigma$ 为单射，则

\begin{equation}
    \exists\,\varphi\in\operatorname{Aut}(H),\ \varphi(G_{\sigma_1})=G_{\sigma_2}
    \quad\Longleftrightarrow\quad
    \sigma_2=\varphi\sigma_1\varphi^{-1}.
\end{equation}

\textit{证明。}由式~\eqref{eq:sym-A1}，
$\varphi(G_{\sigma_1})=L(H,\varphi\sigma_1\varphi^{-1})=G_{\varphi\sigma_1\varphi^{-1}}$；由单射性即得。$\square$

\textbf{定理 2。}
在引理 2 条件下，等价类恰为 $\operatorname{Aut}(H)$ 在 $S_n$ 上的共轭轨道：

\begin{equation}
    \sigma_1 \sim \sigma_2
    \quad\Longleftrightarrow\quad
    \exists\,\varphi\in\operatorname{Aut}(H),\ \sigma_2=\varphi\sigma_1\varphi^{-1}.
\end{equation}

\textbf{注。}共轭不是额外假设——它是「在对称图上做节点重标号」这一操作的代数表达。
$L$ 的单射性若放宽，等价只会变粗，共轭结构不变。

\section{轨道计数}

\textbf{命题 1（轨道大小）。}对 $\sigma\in S_n$，其在 $\operatorname{Aut}(H)$ 共轭下的轨道大小为

\begin{equation}
    |\operatorname{orb}(\sigma)| \;=\;
    \frac{|\operatorname{Aut}(H)|}{|\operatorname{Aut}(H)\cap C_{S_n}(\sigma)|},
\end{equation}

其中 $C_{S_n}(\sigma)=\{\tau\in S_n:\tau\sigma=\sigma\tau\}$ 为 $\sigma$ 在 $S_n$ 中的中心化子。

\textbf{命题 2（共轭类内轨道数）。}
对 cycle type $\lambda\vdash n$，该 $S_n$ 共轭类内的 $G$-轨道数（$G=\operatorname{Aut}(H)$）为

\begin{equation}
    N_\lambda(G) \;=\; \frac{n!}{|G|}\cdot\frac{|G\cap C_{S_n}(\sigma_\lambda)|}{|C_{S_n}(\sigma_\lambda)|},
\end{equation}

其中 $\sigma_\lambda$ 为 cycle type $\lambda$ 的任一代表元。

\textbf{命题 3（总轨道数，Burnside 引理）。}

\begin{equation}
    \text{总轨道数} \;=\; \frac{1}{|G|}\sum_{g\in G} |C_{S_n}(g)|,
    \qquad
    |C_{S_n}(g)| \;=\; \prod_{k=1}^{n} k^{m_k}\cdot m_k!,
\end{equation}

其中 $g$ 的 cycle type 为 $1^{m_1}2^{m_2}\cdots n^{m_n}$。

\section{常见拓扑的自同构群}

\begin{center}
\begin{tabular}{l|l|l}
\hline
物理拓扑 $H$ & $\operatorname{Aut}(H)$ & 阶 \\
\hline
$K_n$（完全图） & $S_n$ & $n!$ \\
$C_n$（环） & $D_n$（二面体群） & $2n$ \\
$\text{Mesh}(k)$（$k\times k$ 网格） & $D_4$ & 8 \\
$\text{Torus}(k)$ & $(C_k\times C_k)\rtimes C_2$ & $2k^2$ \\
$\text{Dragonfly}(a,p,h)$ & 取决于布线，正则布线时接近 $S_a\wr S_p$ & — \\
$Q_d$（超立方，$n=2^d$） & 超正八面体群 & $d!\cdot 2^d$ \\
$K_{m,m}$（完全二分图） & $S_m\times S_m\rtimes C_2$ & $2(m!)^2$ \\
\hline
\end{tabular}
\end{center}

当 $G=S_n$ 时，轨道即 $S_n$ 共轭类，对应 $n$ 的整数分拆，共 $p(n)$ 个。

\section{可行性}

对 $n\le 10$，分拆数 $p(n)\le 42$。$n$ 增至 12--15 时 $p(12)=77$、$p(15)=176$，仍可控；
但若 $\operatorname{Aut}(H)$ 退化为平凡群 $\{e\}$，轨道数退化为 $15!\approx 1.3\times 10^{12}$，
需更强的归约或启发式。对于本文涉及的对称拓扑（Mesh/Torus/Dragonfly），
$\operatorname{Aut}(H)$ 非平凡，$|\mathcal{R}|$ 远小于 $n!$，是第~\ref{sec:scalability}节
「LP 规模受控」的依据。

\chapter{热约束的线性化：M-矩阵与保序消元}\label{app:thermal}

本附录给出第~\ref{sec:ov-linearize}节热约束消元（$\mathbf{G}^{-1}\ge 0$）的完整推导，
对应第~\ref{sec:ov-thermal}节的稳态热网络式~\eqref{eq:thermal}。

\section{热网络的 M-矩阵结构}

热传导与电路同构：温差 $\Delta T$ 对应电压、热流 $P$ 对应电流、热阻 $R_{\text{th}}$ 对应电阻，
欧姆定律的热学版本为 $P=\Delta T/R_{\text{th}}$。为书写方便用热导 $g=1/R_{\text{th}}$。

把每个 interposer 聚合为一个热节点，节点 $i$ 的稳态热平衡为
「产热 = 向上散热 + 向邻居横向散热」：

\begin{equation}\label{eq:th-node}
    P_i \;=\; g_{\text{vert}}\bigl(T_i-T_{\text{amb}}\bigr)
    \;+\; \sum_{j\in\mathcal{N}(i)} g_{\text{lat}}\bigl(T_i-T_j\bigr),
\end{equation}

其中 $g_{\text{vert}}=1/R_{\text{th,vert}}$ 为 die $\to$ 环境的垂直热导（TIM + lid + heatsink 集总），
$g_{\text{lat}}=1/R_{\text{th,lat}}$ 为相邻节点经 substrate 的横向热导，$T_{\text{amb}}$ 为环境温度。
整理为矩阵形式：

\begin{equation}\label{eq:th-matrix}
    \mathbf{G}\cdot\mathbf{T} \;=\; \mathbf{P} + \mathbf{b},
\end{equation}

其中

\begin{equation}
    G_{ii}=g_{\text{vert}}+|\mathcal{N}(i)|\,g_{\text{lat}},
    \qquad
    G_{ij}=-g_{\text{lat}}\ (j\in\mathcal{N}(i)),
    \qquad
    b_i=g_{\text{vert}}\,T_{\text{amb}}.
\end{equation}

$\mathbf{G}$ 是对角占优的\textbf{M-矩阵}：对角元全正、非对角元非正、
每行对角元不小于非对角元绝对值之和（$G_{ii}\ge\sum_{j\neq i}|G_{ij}|$，此处严格大于，差即 $g_{\text{vert}}$），
且稀疏（每节点最多连 4 个邻居）、对称正定。

\section{M-矩阵的逆非负性}

M-矩阵的核心性质是\textbf{逆非负}：$\mathbf{G}^{-1}\ge 0$（逐元素非负）~\cite{berman1994nonnegative}。
其物理含义是热传导的因果性：

\begin{quote}
任何一个节点的功率增加，所有节点的温度只升不降；功率减少，温度只降不升。
\end{quote}

即 $\mathbf{T}=\mathbf{G}^{-1}(\mathbf{P}+\mathbf{b})$ 是 $\mathbf{P}$ 的\textbf{单调}（保序）映射——
这正是第~\ref{sec:ov-linearize}节能把温度约束线性化、且保证消元后不等式方向不变的原因。

\section{温度约束的线性化}

要求所有节点温度不超过结温上限 $T_{\max}$：

\begin{equation}\label{eq:th-ineq}
    \mathbf{T} \;=\; \mathbf{G}^{-1}(\mathbf{P}+\mathbf{b}) \;\le\; T_{\max}\cdot\mathbf{1}.
\end{equation}

因 $\mathbf{G}^{-1}$ 为常数矩阵，式~\eqref{eq:th-ineq} 已是关于 $\mathbf{P}$ 的线性不等式。
代入功耗模型 $\mathbf{P}=\mathbf{P}_{peak}(B)\cdot\mathbf{1}+\mathbf{M}\mathbf{S}_{\text{dyn}}\boldsymbol{\ell}$
与 lane 定义 $\boldsymbol{\ell}=B\,\mathbf{S}_{\text{bw}}^{-1}\mathbf{L}$（式~\eqref{eq:lane-def}），
整理得\textbf{关于 $\mathbf{L}$ 的线性不等式}：

\begin{equation}\label{eq:th-linearized}
    B\cdot\underbrace{\mathbf{G}^{-1}\mathbf{M}\mathbf{S}_{\text{dyn}}\mathbf{S}_{\text{bw}}^{-1}}_{\mathbf{K}}
    \cdot\mathbf{L}
    \;\le\;
    \underbrace{T_{\max}\cdot\mathbf{1}-\mathbf{G}^{-1}\bigl(\mathbf{P}_{peak}(B)\cdot\mathbf{1}+\mathbf{b}\bigr)}_{\mathbf{rhs}}.
\end{equation}

固定 $B$ 后，系数矩阵 $\mathbf{K}$ 与右手边 $\mathbf{rhs}$ 均为常数，问题保持线性——
这正是第~\ref{sec:ov-unified}节统一 LP 中热约束的消元形式。
$\mathbf{K}\ge 0$（因 $\mathbf{G}^{-1}\ge 0$、$\mathbf{M},\mathbf{S}_{\text{dyn}},\mathbf{S}_{\text{bw}}^{-1}$ 均非负）
保证「负载越重、左侧越大」，物理单调性贯穿消元全过程。

\section{L0 粗筛（全局功耗）}

热 L0 是上述消元的进一步粗化：把整个 interposer 聚为单热节点、忽略温度梯度，
只约束总功耗不超过面积热密度上限：

\begin{equation}\label{eq:th-l0}
    \mathbf{1}^T\mathbf{P} \;\le\; A_{ip}\cdot q_{\max},
\end{equation}

其中 $A_{ip}$ 为 interposer 面积、$q_{\max}$ 为散热密度上限。
它比式~\eqref{eq:th-linearized} 更保守（忽略横向热扩散、只留垂直散热），
与 L1 的判定一致性在第~\ref{sec:fidelity}节讨论。
